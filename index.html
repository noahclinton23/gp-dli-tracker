// ========================= // REPORT HELPERS // ========================= function getEv(ev, key) { return ev[key] || 0; } function gradeFromPer10(per10, defPoss, hasEvents) { if (!defPoss && !hasEvents) return 'N/A'; if (per10 >= 3.0) return 'A+'; if (per10 >= 2.5) return 'A'; if (per10 >= 2.0) return 'A-'; if (per10 >= 1.5) return 'B+'; if (per10 >= 1.0) return 'B'; if (per10 >= 0.5) return 'B-'; if (per10 >= 0.0) return 'C+'; if (per10 >= -0.5) return 'C'; if (per10 >= -1.0) return 'C-'; if (per10 >= -1.5) return 'D+'; if (per10 >= -2.0) return 'D'; return 'F'; } function gradeClass(letter) { if (letter === 'N/A') return ''; const base = letter[0]; // A,B,C,D,F if (base === 'A') return 'grade-A'; if (base === 'B') return 'grade-B'; if (base === 'C') return 'grade-C'; if (base === 'D') return 'grade-D'; return 'grade-F'; } // Build report view container in DOM function initReportView() { const reportDiv = document.createElement("div"); reportDiv.id = "reportView"; reportDiv.className = "container"; reportDiv.style.display = "none"; reportDiv.innerHTML = `
