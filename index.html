// ========================= // POST GAME REPORT // ========================= function grade(per10){ if (per10 === "" || per10 === null) return "N/A"; const v = parseFloat(per10); if (v >= 3.0) return "A"; if (v >= 2.0) return "B"; if (v >= 1.0) return "C"; if (v >= 0.0) return "D"; return "F"; } function strengthsNeeds(ev){ const s = []; const n = []; if ((ev.STL||0) >= 2) s.push("Steals"); if ((ev.STLS||0) >= 1) s.push("Steal→Score"); if ((ev.DEF||0) >= 2) s.push("Deflections"); if ((ev.BXO||0) >= 2) s.push("Box Outs"); if ((ev.TR||0) >= 3) s.push("Team Rebounds"); if ((ev.BXOR||0) >= 1) s.push("Own Rebounds"); if ((ev.WU||0) >= 1) s.push("Wall Ups"); if ((ev.SB||0) >= 2) s.push("Sprint Back"); if ((ev.CH||0) >= 1) s.push("Charges"); if ((ev.DLB||0) >= 1) s.push("Loose Balls"); if ((ev.EH||0) >= 2) s.push("Early Help"); if ((ev.BB||0) >= 1) n.push("Blow-Bys"); if ((ev.F||0) >= 2) n.push("Fouls"); if ((ev.GAM||0) >= 1) n.push("Gambles"); if ((ev.GAMS||0) >= 1) n.push("Gambles→Score"); return { s: s.length? s.join(", ") : "No major strengths", n: n.length? n.join(", ") : "No major issues" }; } function renderReport(){ const body = document.getElementById("reportSummaryBody"); body.innerHTML = ""; players.forEach(p => { const st = state[p.id]; const ev = st.events || {}; let raw = 0; for (let k in ev) raw += ev[k]*(weightMap[k]||0); const per10 = st.dp>0 ? (raw/st.dp*10).toFixed(2) : ""; const g = grade(per10); const sn = strengthsNeeds(ev); const tr = document.createElement("tr"); function td(v, cls){ const c=document.createElement("td"); c.textContent=v; if(cls) c.className=cls; return c; } tr.appendChild(td(`#${p.number} ${p.name}`)); tr.appendChild(td(st.dp)); tr.appendChild(td(raw)); tr.appendChild(td(per10)); tr.appendChild(td(g, `grade-${g}`)); tr.appendChild(td(sn.s)); tr.appendChild(td(sn.n)); body.appendChild(tr); }); } // ========================= // TAB SWITCHING // ========================= document.getElementById("tabLive").addEventListener("click", () => { document.getElementById("tabLive").classList.add("active"); document.getElementById("tabReport").classList.remove("active"); document.getElementById("liveView").style.display = "block"; document.getElementById("reportView").style.display = "none"; }); document.getElementById("tabReport").addEventListener("click", () => { document.getElementById("tabReport").classList.add("active"); document.getElementById("tabLive").classList.remove("active"); document.getElementById("liveView").style.display = "none"; renderReport(); document.getElementById("reportView").style.display = "block"; }); // ========================= // INITIAL DRAW // ========================= renderPlayers(); renderEventButtons(); renderSummary();
